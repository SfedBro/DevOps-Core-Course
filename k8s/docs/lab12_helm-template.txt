# Source: mychart/templates/hooks/post-install-job.yaml
apiVersion: batch/v1
kind: Job
metadata:
  name: "lab12-mychart-post-install"
  labels:
    helm.sh/chart: mychart-0.1.0
    app.kubernetes.io/name: mychart
    app.kubernetes.io/instance: lab12
    app.kubernetes.io/version: "latest"
    app.kubernetes.io/managed-by: Helm
    app.kubernetes.io/part-of: devops-core-course
  annotations:
    "helm.sh/hook": post-install
    "helm.sh/hook-weight": "5"
    "helm.sh/hook-delete-policy": before-hook-creation,hook-succeeded
spec:
  template:
    metadata:
      name: "lab12-mychart-post-install"
      labels:
        app.kubernetes.io/name: mychart
        app.kubernetes.io/instance: lab12
    spec:
      restartPolicy: Never
      containers:
        - name: post-install-smoke-test
          image: "busybox:1.36"
          imagePullPolicy: IfNotPresent
          command:
            - sh
            - -c
            - |
              set -eu
              i=0
              until wget -qO- "http://lab12-mychart:80/health" >/dev/null 2>&1; do
                i=$((i+1))
                if [ "$i" -ge 15 ]; then
                  echo "Smoke test failed for lab12-mychart"
                  exit 1
                fi
                echo "Waiting for application to become reachable..."
                sleep 2
              done
              echo "Post-install smoke test passed for lab12-mychart"
  backoffLimit: 0
---
